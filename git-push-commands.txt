# Step 1: Navigate to project

# Step 2: Initialize Git
git init

# Step 3: Add all files
git add .

# Step 4: Commit
git commit -m "Initial commit: Real Estate Platform Backend with Subscription System"

# Step 5: Add remote (REPLACE with YOUR GitHub repo URL)
git remote add origin https://github.com/YOUR-USERNAME/YOUR-REPO-NAME.git

# Step 6: Verify remote
git remote -v

# Step 7: Push to GitHub
git push -u origin master

# =====================================================
# If you get error: "failed to push some refs"
# =====================================================
git pull origin master --allow-unrelated-histories
git push -u origin master

# =====================================================
# If branch name is "main" instead of "master"
# =====================================================
git branch -M main
git push -u origin main
